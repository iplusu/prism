```python
import subprocess
import os


def count_total_lines(dirname):
    result = subprocess.run(
        ["wc", "-l"],
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        cwd=dirname
    )
    # If no files, wc -l with no args reads from stdin (DEVNULL) and outputs 0
    # But we want to count all text files in the directory
    # Let's use find + xargs or a different approach
    
    # Actually, let's use a shell command that finds all files and counts lines
    result = subprocess.run(
        "find . -type f -print0 | xargs -0 wc -l 2>/dev/null | tail -1",
        shell=True,
        cwd=dirname,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True
    )
    output = result.stdout.strip()
    if not output:
        return 0
    # The last line of wc -l output is the total
    # Format: "total filename1 filename2 ..."
    parts = output.split()
    if parts:
        try:
            return int(parts[0])
        except ValueError:
            return 0
    return 0
```
