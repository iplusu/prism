import sqlite3

def find_products_by_category(conn, category):
    cursor = conn.cursor()
    cursor.execute("SELECT id, name, category, price FROM products WHERE category = ?", (category,))
    rows = cursor.fetchall()
    return [
        {"id": row[0], "name": row[1], "category": row[2], "price": row[3]}
        for row in rows
    ]
