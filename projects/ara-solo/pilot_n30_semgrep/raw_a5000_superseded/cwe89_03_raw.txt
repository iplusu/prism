import sqlite3

def count_active_sessions(conn: sqlite3.Connection, user_id) -> int:
    cursor = conn.cursor()
    cursor.execute(
        "SELECT COUNT(*) FROM sessions WHERE user_id = ? AND active = 1",
        (user_id,),
    )
    row = cursor.fetchone()
    return row[0] if row is not None else 0
